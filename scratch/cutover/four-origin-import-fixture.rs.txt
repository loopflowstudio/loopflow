//! Actual CLI acceptance over a populated historical Home. No provider is launched.
#![allow(dead_code)]

mod support;

use std::fs;
use std::path::{Path, PathBuf};
use std::process::{Command, Output};

use loopflow::durable::{FlowPosition, QueuedInvocation, RunId};
use loopflow::engine::{ConcreteSkill, ConcreteStep, ExecutionCursor, OccurrencePolicy, Skill};
use loopflow_test_support::TestRepo;
use serde_json::{json, Value};

fn command(home: &Path, repo: &Path, args: &[&str]) -> Output {
    let mut cmd = Command::new(env!("CARGO_BIN_EXE_lf"));
    for (key, _) in std::env::vars_os() {
        if key.to_string_lossy().starts_with("LF_")
            || key.to_string_lossy().starts_with("LOOPFLOW_")
        {
            cmd.env_remove(key);
        }
    }
    cmd.args(args)
        .current_dir(repo)
        .env("LF_HOME", home)
        .env("LF_CONTROL_HOME", home)
        .env("LF_DB_PATH", home.join("loopflow.db"))
        .env("LF_CONTROL_DB_PATH", home.join("loopflow.db"))
        .env("LF_BIN", env!("CARGO_BIN_EXE_lf"))
        .env("RUST_LOG", "off")
        .output()
        .unwrap()
}

fn cli(home: &Path, repo: &Path, args: &[&str]) -> Value {
    let output = command(home, repo, args);
    assert!(output.status.success(), "{args:?}: {}", String::from_utf8_lossy(&output.stderr));
    serde_json::from_slice(&output.stdout).unwrap()
}

fn write_json(path: &Path, value: &Value) {
    fs::create_dir_all(path.parent().unwrap()).unwrap();
    fs::write(path, serde_json::to_vec_pretty(value).unwrap()).unwrap();
}

fn run_dir(home: &Path, id: &str) -> PathBuf {
    home.join("runs").join(&id[4..6]).join(id)
}

struct FourOrigins {
    home: tempfile::TempDir,
    repo: TestRepo,
    interactive: String,
    ask: String,
    task_review: String,
    taskless_review: String,
    task_id: String,
    wave_id: String,
    invocation: String,
    capture: Value,
    retired: Vec<PathBuf>,
    manifests: Vec<(PathBuf, Vec<u8>)>,
}

impl FourOrigins {
    fn seed() -> Self {
        let home = tempfile::tempdir().unwrap();
        let repo = TestRepo::new();
        let task = support::register_unrun_task(home.path(), repo.path(), "session-cutover", &repo.head_sha());
        let step = ConcreteStep::Skill(ConcreteSkill {
            skill: Skill { content: Some("Captured review; the source is unavailable.".into()), ..Skill::named("review-design") },
            policy: OccurrencePolicy { id: Some("review".into()), human: true, ..Default::default() },
            flow_parents: vec![],
        });
        let task_run = RunId::new();
        let position = tokio::runtime::Runtime::new().unwrap().block_on(task.store.set_flow_position(
            &task.task.id,
            FlowPosition {
                task_id: task.task.id.clone(),
                invocation: QueuedInvocation::new("captured-task", vec![step.clone()]).unwrap(),
                session_run_id: Some(task_run.clone()),
                ready_summary: Some("Task feedback retained".into()),
                cursor: ExecutionCursor::default(), version: 0, worker_generation: 0,
                claim: None, failure: None, updated_at: time::OffsetDateTime::now_utc(),
            },
        )).unwrap();
        let task_review = format!("{}:{}:captured-task:review:0", task.task.id, position.invocation.id);
        let interactive = RunId::new().to_string();
        let ask_run = RunId::new().to_string();
        let taskless_run = RunId::new().to_string();
        let invocation = uuid::Uuid::new_v4().to_string();
        let cursor = ExecutionCursor::default();
        let boundary = cursor.boundary_key();
        let taskless_review = format!("flow:{invocation}:{boundary}");
        let ask = "ask_completed-keyed-answer".to_string();
        let mut manifests = Vec::new();
        let mut retired = Vec::new();
        for (id, title, subjects) in [
            (interactive.as_str(), "Interactive human name", json!([{ "selector": format!("task:{}", task.task.id), "source": "declared" }])),
            (ask_run.as_str(), "Answered question", json!([])),
            (taskless_run.as_str(), "Taskless human name", json!([])),
            (task_run.as_str(), "Task review human name", json!([])),
        ] {
            let dir = run_dir(home.path(), id);
            let manifest = json!({
                "schema_version":1, "run_id":id, "parent_run_id":null,
                "created_at":"2026-09-26T20:00:00Z", "harness":"codex", "model":null,
                "surface":"tui", "cwd":repo.path(), "repo":repo.path(), "worktree":repo.path(),
                "skill":null, "subjects":subjects, "flow":null, "launch":null,
                "context":null, "runtime_path":null, "runtime_digest":null, "host":"fixture", "boot_id":null
            });
            write_json(&dir.join("manifest.json"), &manifest);
            manifests.push((dir.join("manifest.json"), fs::read(dir.join("manifest.json")).unwrap()));
            write_json(&dir.join("session-name.json"), &json!({"schema_version":1,"title":title,"source":"human"}));
            retired.push(dir.join("session-name.json"));
            write_json(&dir.join("provider-session.json"), &json!({"schema_version":1,"provider_session_id":format!("history-{id}"),"account_id":null}));
            write_json(&dir.join("terminal.json"), &json!({"schema_version":1,"outcome":"completed","ended_at":"2026-09-26T20:05:00Z","result_ref":null}));
        }
        let ask_path = home.path().join("human-sessions").join(format!("{ask}.json"));
        write_json(&ask_path, &json!({
            "id":ask,"parent_run_id":interactive,"parent_run_dir":run_dir(home.path(), &interactive),
            "work":null,"work_selector":null,"title":"Answered question","detail":"fixture",
            "prompt":"Keep the answer","skill":null,"cwd":repo.path(),"model":"codex",
            "session_run_id":ask_run,"ready_summary":"Completed keyed answer",
            "status":{"completed":{"summary":"Completed keyed answer"}},"retain_completed":true
        }));
        retired.push(ask_path);
        let capture = json!({
            "id":invocation,"flow":"deleted-template","cwd":repo.path(),"steps":[step],
            "cursor":cursor,"message":null,"model":"codex","wave":null,"task":null,"as_work":null,
            "active":{"id":boundary,"run_id":taskless_run,"run_dir":run_dir(home.path(), &taskless_run),"completed":false,"ready_summary":"Taskless feedback retained"},
            "failure":null,"finished":false
        });
        let flow_path = home.path().join("flows").join(&invocation).join("position.json");
        write_json(&flow_path, &capture);
        retired.push(flow_path);
        Self { home, repo, interactive, ask, task_review, taskless_review,
            task_id:task.task.id.to_string(), wave_id:task.task.wave_id.to_string(),
            invocation, capture, retired, manifests }
    }

    fn import(&self) -> Value {
        cli(self.home.path(), self.repo.path(), &["session","import","--from",self.home.path().to_str().unwrap(),"--json"])
    }
}

#[test]
fn four_origin_import_retains_conversations_without_file_authority() {
    let fixture = FourOrigins::seed();
    fixture.import();
    fixture.import(); // Same source is a no-op, including completed keyed answers.
    let conn = rusqlite::Connection::open(fixture.home.path().join("loopflow.db")).unwrap();
    assert_eq!(conn.query_row("SELECT count(*) FROM sessions", [], |r| r.get::<_,i64>(0)).unwrap(), 4);
    assert_eq!(conn.query_row("SELECT count(*) FROM runs", [], |r| r.get::<_,i64>(0)).unwrap(), 4);
    for path in &fixture.retired { fs::remove_file(path).unwrap(); }
    let listed = cli(fixture.home.path(), fixture.repo.path(), &["session","list","--all","--json"]);
    let sessions = listed.as_array().unwrap();
    assert_eq!(sessions.len(), 3, "completed Ask remains queryable but is not open inventory");
    for (id, title, feedback) in [
        (&fixture.interactive, "Interactive human name", None),
        (&fixture.task_review, "Task review human name", Some("Task feedback retained")),
        (&fixture.taskless_review, "Taskless human name", Some("Taskless feedback retained")),
    ] {
        let session = sessions.iter().find(|s| s["id"] == *id).unwrap();
        assert_eq!(session["title"], title);
        assert_eq!(session["ready_summary"], json!(feedback));
        let renamed = cli(fixture.home.path(), fixture.repo.path(), &["session","rename",id,"Retained conversation","--json"]);
        assert_eq!(renamed["id"], *id);
        assert_eq!(renamed["title"], "Retained conversation");
    }
    for path in &fixture.retired { assert!(!path.exists(), "recreated {}", path.display()); }
    for (path, bytes) in &fixture.manifests { assert_eq!(&fs::read(path).unwrap(), bytes); }
    let parents: (String, String) = conn.query_row("SELECT task_id,wave_id FROM runs WHERE id=?1", [&fixture.interactive], |r| Ok((r.get(0)?,r.get(1)?))).unwrap();
    assert_eq!(parents, (fixture.task_id, fixture.wave_id));
    let answer: String = conn.query_row("SELECT ready_summary FROM sessions WHERE id=?1 AND completed_at IS NOT NULL", [&fixture.ask], |r| r.get(0)).unwrap();
    assert_eq!(answer, "Completed keyed answer");
}
