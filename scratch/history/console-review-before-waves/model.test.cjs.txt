const {test}=require('node:test');const assert=require('node:assert/strict');const {create,visit,travel,matches,pin}=require('./model.js');
test('repository windows keep independent selections and drafts while search spans repositories',()=>{const {allTasks}=require('./model.js');const a=create(),b=create({},'fieldnotes');a.drafts['489']='Keep this here';visit(b,{kind:'task',id:'102'});assert.equal(a.route.id,'489');assert.equal(a.drafts['489'],'Keep this here');assert.deepEqual(b.drafts,{});assert.deepEqual(matches('repo:fieldnotes state:available',allTasks).map(t=>t.key),['FN-101']);});
test('switching Tasks keeps working-set positions; opening new work adds it once',()=>{const s=create();const before=[...s.workingSet];visit(s,{kind:'task',id:'416'});visit(s,{kind:'task',id:'489'});assert.deepEqual(s.workingSet,before);visit(s,{kind:'task',id:'504'});visit(s,{kind:'task',id:'416'});visit(s,{kind:'task',id:'504'});assert.deepEqual(s.workingSet,[...before,'504']);});
test('Task visits preserve the source scope and draft through Back/Forward',()=>{const s=create();visit(s,{kind:'wave',id:'product'});visit(s,{kind:'task',id:'489'});s.drafts['489']='Keep the sidebar stable';visit(s,{kind:'wave',id:'infrastructure'});travel(s);assert.deepEqual(s.route,{kind:'task',id:'489'});assert.equal(s.scope,'product');assert.equal(s.drafts['489'],'Keep the sidebar stable');travel(s,false);assert.equal(s.scope,'infrastructure')});
test('filters intersect and enumerated alternatives union without duplicate Tasks',()=>{assert.deepEqual(matches('wave:product state:available').map(t=>t.id),['489','499']);assert.deepEqual(matches('task:#489 or task:LOO-416 or task:#489').map(t=>t.id),['489','416']);assert.equal(matches('wave:missing').length,0)});
test('pinning the same filter reuses the saved view without changing Task ownership',()=>{const s=create();const a=pin(s,'Product reviews','wave:product state:available');const b=pin(s,'Reviews','wave:product state:available');assert.equal(a,b);assert.equal(s.views.length,3);assert.deepEqual(matches(a.query).map(t=>t.wave),['product','product'])});
test('a new turn cannot carry the prior turn report into current activity',()=>{const {activityPreview}=require('./model.js');const report={turn:'previous',text:'Tests passed'};assert.deepEqual(activityPreview({phase:'running',observed:true,turn:'next',report}),{label:'Working',text:'Working',tone:'running'});assert.equal(activityPreview({phase:'available',observed:true,turn:'next',report}).text,'Ready for your next step');});
test('lost observation is unknown, while finished reports are explicitly historical',()=>{const {activityPreview}=require('./model.js');const report={turn:'one',text:'Design ready'};assert.equal(activityPreview({phase:'running',observed:false,turn:'one',report}).text,'Activity unavailable');assert.equal(activityPreview({phase:'idle',observed:true,turn:'one',report}).text,'Last result · Design ready');});
test('navigator prioritizes working recency and exact IDs without moving sidebar positions',()=>{const {navigatorTasks}=require('./model.js');const s=create(),positions=[...s.workingSet];visit(s,{kind:'task',id:'505'});assert.equal(navigatorTasks(s,'loopflow','')[0].id,'505');assert.equal(navigatorTasks(s,'loopflow','LOO-416')[0].id,'416');assert.deepEqual(s.workingSet,positions);});
